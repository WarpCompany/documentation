These examples show common Pay with Warp integrations. They are starting points — add your own persistence, error handling and logging before using them in production.

<Warning>
  Create sessions on your server with your secret key, using an amount from your own records. Never send the secret key to the browser and never trust an amount sent by the browser.
</Warning>

<Tip>
  These examples call the REST API with `fetch` and `requests`. In Node.js you can use the [Node.js SDK](/sdks/node) instead, and to show the checkout on your own page see [Embedded checkout](/warp-pay/embedded-checkout).
</Tip>

## Prerequisites

- A Warp project and its secret key in `WARP_SECRET_KEY` (`sk_test_...` while testing).
- A webhook URL and signing secret configured in the Warp Platform (for the webhook examples). See [Webhooks](/warp-pay/webhooks).

## Create a session from an Express backend

This endpoint looks up the order, creates a checkout session and returns the hosted `checkout_url` to the browser.

```javascript server.js
import express from 'express';

const WARP_API = 'https://api.usewarp.net/v1';

const app = express();
app.use('/api', express.json());

app.post('/api/checkout', async (req, res) => {
  const order = await getOrder(req.body.orderId); // your own data layer
  if (!order) return res.status(404).json({ error: 'Order not found' });

  const warpRes = await fetch(`${WARP_API}/checkout/sessions`, {
    method: 'POST',
    headers: {
      Authorization: `Bearer ${process.env.WARP_SECRET_KEY}`,
      'Content-Type': 'application/json',
    },
    body: JSON.stringify({
      amount_usd: order.totalUsd, // e.g. 99.99
      description: `Order #${order.id}`,
      metadata: { order_id: order.id },
      return_url: `https://example.com/orders/${order.id}`,
      cancel_url: 'https://example.com/cart',
    }),
  });

  if (!warpRes.ok) {
    const { error } = await warpRes.json();
    console.error('Warp error:', error.code, error.message);
    return res.status(502).json({ error: 'Could not start checkout' });
  }

  const session = await warpRes.json();
  await saveSessionId(order.id, session.id);
  res.json({ checkoutUrl: session.checkout_url });
});

app.listen(3000);
```

## Add a checkout button in React

The button calls your backend (above), which returns the hosted checkout URL. The secret key never reaches the browser.

```jsx CheckoutButton.jsx
import { useState } from 'react';

export default function CheckoutButton({ orderId }) {
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState(null);

  async function handleCheckout() {
    setLoading(true);
    setError(null);
    try {
      const res = await fetch('/api/checkout', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ orderId }),
      });
      if (!res.ok) throw new Error('Could not start checkout');
      const { checkoutUrl } = await res.json();
      window.location.assign(checkoutUrl);
    } catch (err) {
      setError(err.message);
      setLoading(false);
    }
  }

  return (
    <div>
      <button onClick={handleCheckout} disabled={loading}>
        {loading ? 'Redirecting...' : 'Pay with Warp'}
      </button>
      {error && <p role="alert">{error}</p>}
    </div>
  );
}
```

## Confirm a payment on the return page

Webhooks are the primary signal, but when the customer lands on your `return_url` you can also check the session directly before showing a confirmation.

```javascript
async function getSession(sessionId) {
  const res = await fetch(
    `https://api.usewarp.net/v1/checkout/sessions/${encodeURIComponent(sessionId)}`,
    { headers: { Authorization: `Bearer ${process.env.WARP_SECRET_KEY}` } },
  );
  if (!res.ok) throw new Error(`Warp returned HTTP ${res.status}`);
  return res.json();
}

const session = await getSession(await getSessionIdForOrder(orderId));

switch (session.status) {
  case 'completed':
    // Paid: session.payment has chain, token, tx_hash, amount and confirmations
    break;
  case 'pending':
  case 'processing':
    // Not final yet: show "confirming payment" and wait for the webhook
    break;
  case 'failed':
  case 'expired':
    // Offer to start a new checkout
    break;
}
```

## Create a session and list sessions in Python

```python warp.py
import os

import requests

WARP_API = "https://api.usewarp.net/v1"
HEADERS = {"Authorization": f"Bearer {os.environ['WARP_SECRET_KEY']}"}


def create_session(order_id: str, amount_usd: float) -> dict:
    res = requests.post(
        f"{WARP_API}/checkout/sessions",
        headers=HEADERS,
        json={
            "amount_usd": amount_usd,
            "description": f"Order #{order_id}",
            "metadata": {"order_id": order_id},
            "return_url": f"https://example.com/orders/{order_id}",
            "cancel_url": "https://example.com/cart",
        },
        timeout=10,
    )
    res.raise_for_status()
    return res.json()


def completed_sessions():
    """Yield every completed session, following the pagination cursor."""
    params = {"status": "completed", "limit": 100}
    while True:
        res = requests.get(f"{WARP_API}/checkout/sessions", headers=HEADERS, params=params, timeout=10)
        res.raise_for_status()
        page = res.json()
        yield from page["data"]
        if not page["has_more"]:
            break
        params["starting_after"] = page["data"][-1]["id"]


if __name__ == "__main__":
    session = create_session("1234", 49.99)
    print("Send the customer to:", session["checkout_url"])
```
